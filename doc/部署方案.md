# docker-compose.yml

```yaml
version: "3.9"

networks:
  custom_net:
    driver: bridge

services:
  mysql:
    image: mysql:5.7.44
    container_name: mysql57
    restart: always
    privileged: true
    ports:
      - "8306:3306"
    environment:
      MYSQL_ROOT_PASSWORD: 123456
    volumes:
      - E:/docker/compose/mysql/conf/my.cnf:/etc/mysql/conf.d/my.cnf:ro
      - E:/docker/compose/mysql/data:/var/lib/mysql
      - E:/docker/compose/mysql/log:/var/log/mysql
    networks:
      - custom_net

  redis:
    image: redis:7.4.2
    container_name: redis74
    restart: always
    ports:
      - "6379:6379"
    volumes:
      - E:/docker/compose/redis/data:/data
      - E:/docker/compose/redis/conf/redis.conf:/etc/redis/redis.conf
    command: ["redis-server", "/etc/redis/redis.conf"]
    networks:
      - custom_net

  gkzh:
    image: openjdk:17.0.2
    container_name: gkzh
    restart: always
    working_dir: /app
    volumes:
      - E:/docker/compose/www/backend:/app
    environment:
      - TZ=Asia/Shanghai
    command: ["java", "-jar", "gkzh-admin.jar"]
    ports:
      - "8083:8080"
    networks:
      - custom_net
    depends_on:
      - mysql
      - redis

  nginx:
    image: nginx:1.29.0
    container_name: nginx
    restart: always
    privileged: true
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - E:/docker/compose/www/app:/usr/share/nginx/html/app
      - E:/docker/compose/www/frontend/dist:/usr/share/nginx/html/admin
      - E:/docker/compose/nginx/conf/nginx.conf:/etc/nginx/nginx.conf
      - E:/docker/compose/nginx/conf/conf.d:/etc/nginx/conf.d
      - E:/docker/compose/nginx/log:/var/log/nginx
      - E:/docker/compose/nginx/ssl:/etc/nginx/ssl
    networks:
      - custom_net
    depends_on:
      - gkzh

```

# 使用说明

##  **1. 创建所需目录**

（如果尚未创建）

```sh
# 注意在windows里下面的命令不生效，可切换到wsl环境
mkdir -p E:/docker/compose/{nginx/{conf,log,html,ssl},mysql/{conf,data,log},redis/{conf,data}}
mkdir -p E:/docker/compose/www/{backend/{config,logs,uploadPath},frontend,app}
```

## **2. 创建外部网络**

（如果尚未创建）

```sh
docker network create custom_net
```

##  **3. 准备配置文件**

Nginx 配置：E:/docker/compose/nginx/conf/nginx.conf 

```conf
error_log  /var/log/nginx/error.log notice;
pid        /var/run/nginx.pid;
events {
    worker_connections  1024;
}
http {
    include       /etc/nginx/mime.types;
    default_type  application/octet-stream;

    log_format  main  '$remote_addr - $remote_user [$time_local] "$request" '
                      '$status $body_bytes_sent "$http_referer" '
                      '"$http_user_agent" "$http_x_forwarded_for"';

    access_log  /var/log/nginx/access.log  main;

    sendfile        on;
    #tcp_nopush     on;

    keepalive_timeout  65;

    #gzip  on;

    include /etc/nginx/conf.d/*.conf;
}
```



 E:/docker/compose/nginx/conf/conf.d/default.conf

```conf
server {
    listen       80;
    listen  [::]:80;
    server_name  xxxxx;

    access_log  /var/log/nginx/host.access.log  main;
    # 上传文件限制
    client_max_body_size 2M;
    location / {
        root /usr/share/nginx/html/app;
        try_files $uri $uri/ /index.html;
    }
    location /admin {
        alias /usr/share/nginx/html/admin/;
        try_files $uri $uri/ /admin/index.html;
    }
    # 专门处理静态资源
    location /admin/static/ {
        alias /usr/share/nginx/html/admin/static/;
        expires 1y;
        add_header Cache-Control "public, immutable";
        add_header Access-Control-Allow-Origin *;
    }
    location /profile/ {
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_pass http://gkzh:8080/profile/;
    }
    # 处理其他静态资源路径
    location /styles/ {
        alias /usr/share/nginx/html/admin/styles/;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    location /prod-api/ {
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_pass http://gkzh:8080/;
        proxy_read_timeout 600s;
        proxy_connect_timeout 600s;
        proxy_send_timeout 600s;
    }
    location /api/ {
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass http://gkzh:8080/api/;
        proxy_read_timeout 600s;
        proxy_connect_timeout 600s;
        proxy_send_timeout 600s;
    }

    # 安全增强
    add_header X-Content-Type-Options nosniff;
    add_header X-Frame-Options SAMEORIGIN;
    add_header X-XSS-Protection "1; mode=block";

    error_page   500 502 503 504  /50x.html;
    location = /50x.html {
        root   /usr/share/nginx/html;
    }
}
```

MySQL 配置：E:/docker/compose/mysql/conf/my.cnf

```
[client]
default-character-set=utf8mb4
 
[mysql]
default-character-set=utf8mb4
 
[mysqld]
default-time-zone = "Asia/Shanghai"
init_connect="SET collation_connection = utf8mb4_unicode_ci"
init_connect="SET NAMES utf8mb4"
character-set-server=utf8mb4
collation-server=utf8mb4_unicode_ci
skip-character-set-client-handshake
skip-name-resolve
default-storage-engine=INNODB
#允许最大连接数
max_connections=1000
#允许连接失败的次数
max_connect_errors=10
```



Redis 配置：E:/docker/compose/redis/conf/redis.conf

```conf
appendonly yes
```

Java 应用配置：E:/docker/compose/www/backend/config/application.yml

```yaml
server:
	port: 8080
spring:
    datasource:
        druid:
            master:
                url: jdbc:mysql://mysql57:3306/gkzh?useUnicode=true&characterEncoding=utf8&zeroDateTimeBehavior=convertToNull&useSSL=true&serverTimezone=GMT%2B8
                username: root
                password: 123456
    redis:
    	host: redis74
    	port: 6379
```

## **4. 启动所有服务**

bash

```shell
docker-compose up -d
# 查看状态
docker ps

# 验证接口是否通了
# 宿主机
curl 127.0.0.1:8083/captchaImage

# 容器内
curl 127.0.0.1:8080/captchaImage
```



## **5. 停止所有服务**

bash

```
docker-compose down
```
